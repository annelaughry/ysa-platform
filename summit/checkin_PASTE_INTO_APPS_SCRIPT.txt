<script>
  // Shared staff access code. Change before the summit.
  const STAFF_PASSPHRASE = "32KMPAYLOAD";
  (function () {
    const gate = document.getElementById('passGate');
    if (sessionStorage.getItem('ysa_staff_ok') === 'true') gate.classList.add('hidden');
    window.checkPassphrase = function () {
      const val = (document.getElementById('passInput').value || '').trim().toUpperCase();
      if (val === STAFF_PASSPHRASE.toUpperCase()) {
        sessionStorage.setItem('ysa_staff_ok', 'true');
        gate.classList.add('hidden');
      } else {
        document.getElementById('passError').style.display = 'block';
      }
    };
  })();

  let currentTab = 'day1';
  let currentFilters = { day1: 'all', day2: 'all', workshop: 'all' };
  let allData = { day1: [], day2: [], workshop: [], all: [] };
  let searchMode = false;
  let searchResults = [];

  function switchTab(tab) {
    currentTab = tab;
    ['day1','day2','workshop','all','trash'].forEach(t => {
      document.getElementById('tab-' + t).classList.toggle('active', t === tab);
      document.getElementById(t + '-panel').style.display = t === tab ? 'block' : 'none';
    });
    if (tab === 'trash') loadTrash();
    else renderCurrentTab();
  }

  function setFilter(day, filter, btn) {
    currentFilters[day] = filter;
    btn.parentElement.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    renderCurrentTab();
  }

  function loadAll(cb) {
    google.script.run.withSuccessHandler(function (items) {
      allData.all = items;
      allData.day1 = items.filter(a => a.attendsDay1);
      allData.day2 = items.filter(a => a.attendsDay2);
      allData.workshop = items.filter(a => a.attendsWorkshop);
      updateStats();
      document.getElementById('refresh-text').textContent = 'Updated ' + new Date().toLocaleTimeString();
      if (cb) cb();
    }).withFailureHandler(handleError).getAllAttendees();
  }

  function updateStats() {
    const list = currentTab === 'all' ? allData.all : (allData[currentTab] || allData.all);
    const ciField = currentTab === 'day1' ? 'day1CheckedIn' : currentTab === 'day2' ? 'day2CheckedIn' : 'workshopCheckedIn';
    const total = list.length;
    const checked = currentTab === 'all' ? list.filter(a => a.day1CheckedIn === 'Yes' || a.day2CheckedIn === 'Yes' || a.workshopCheckedIn === 'Yes').length
                                          : list.filter(a => a[ciField] === 'Yes').length;
    document.getElementById('stat-total').textContent = total;
    document.getElementById('stat-checked').textContent = checked;
    document.getElementById('stat-remaining').textContent = total - checked;
  }

  function renderCurrentTab() {
    updateStats();
    if (searchMode) { renderSearchResults(); return; }
    if (currentTab === 'all') { renderList('all-list', allData.all, 'all'); return; }
    if (currentTab === 'trash') return;
    const filter = currentFilters[currentTab];
    const ciField = currentTab === 'day1' ? 'day1CheckedIn' : currentTab === 'day2' ? 'day2CheckedIn' : 'workshopCheckedIn';
    let list = allData[currentTab] || [];
    if (filter === 'checked') list = list.filter(a => a[ciField] === 'Yes');
    if (filter === 'pending') list = list.filter(a => a[ciField] !== 'Yes');
    renderList(currentTab + '-list', list, currentTab);
  }

  function cardHtml(a, dayContext) {
    const day1Badge = a.attendsDay1 ? `<span class="badge ${a.day1CheckedIn === 'Yes' ? 'badge-checked' : 'badge-pending'}">Day 1: ${a.day1CheckedIn === 'Yes' ? 'Checked In' : 'Not Yet'}</span>` : '';
    const day2Badge = a.attendsDay2 ? `<span class="badge ${a.day2CheckedIn === 'Yes' ? 'badge-checked' : 'badge-pending'}">Day 2: ${a.day2CheckedIn === 'Yes' ? 'Checked In' : 'Not Yet'}</span>` : '';
    const workshopBadge = a.attendsWorkshop ? `<span class="badge ${a.workshopCheckedIn === 'Yes' ? 'badge-checked' : 'badge-pending'}">Workshop: ${a.workshopCheckedIn === 'Yes' ? 'Checked In' : 'Not Yet'}</span>` : '';
    const walkinBadge = (a.confirmNum || '').startsWith('YSA-W') ? '<span class="badge badge-walkin">Walk-in</span>' : '';
    const maybeBadge = (a.attendingStatus || '').includes('maybe') ? '<span class="badge badge-maybe">Maybe RSVP</span>' : '';

    let cardClass = 'result-card';
    const relevantCi = dayContext === 'day1' ? a.day1CheckedIn : dayContext === 'day2' ? a.day2CheckedIn : dayContext === 'workshop' ? a.workshopCheckedIn : null;
    if (relevantCi === 'Yes') cardClass += ' checked-all';

    let actions = '';
    if (dayContext === 'day1' || (dayContext === 'all' && a.attendsDay1)) {
      actions += a.day1CheckedIn === 'Yes'
        ? `<button class="btn btn-amber btn-sm" onclick="doUncheckIn(${a.rowIndex},'day1',this)">Undo Day 1</button>`
        : `<button class="btn btn-green btn-sm" onclick="doCheckIn(${a.rowIndex},'day1',this)">Check In Day 1</button>`;
    }
    if (dayContext === 'day2' || (dayContext === 'all' && a.attendsDay2)) {
      actions += a.day2CheckedIn === 'Yes'
        ? `<button class="btn btn-amber btn-sm" onclick="doUncheckIn(${a.rowIndex},'day2',this)">Undo Day 2</button>`
        : `<button class="btn btn-green btn-sm" onclick="doCheckIn(${a.rowIndex},'day2',this)">Check In Day 2</button>`;
    }
    if (dayContext === 'workshop' || (dayContext === 'all' && a.attendsWorkshop)) {
      actions += a.workshopCheckedIn === 'Yes'
        ? `<button class="btn btn-amber btn-sm" onclick="doUncheckIn(${a.rowIndex},'workshop',this)">Undo Workshop</button>`
        : `<button class="btn btn-green btn-sm" onclick="doCheckIn(${a.rowIndex},'workshop',this)">Check In Workshop</button>`;
    }
    if (a.email) actions += `<button class="btn btn-ghost btn-sm" onclick="doResend(${a.rowIndex},this)">&#128231; Resend Pass</button>`;
    actions += `<button class="btn btn-red btn-sm" onclick="doDelete(${a.rowIndex},'${(a.name||'').replace(/'/g,"\\'")}',this)">Delete</button>`;

    return `
      <div class="result-card ${cardClass.replace('result-card','').trim()}" id="card-${a.rowIndex}">
        <div class="card-top">
          <div>
            <div class="card-name">${a.name}</div>
            ${a.confirmNum ? `<div class="card-conf">${a.confirmNum}</div>` : ''}
          </div>
          <div class="badges">${maybeBadge}${day1Badge}${day2Badge}${workshopBadge}${walkinBadge}<span class="badge badge-role">${a.role || 'Attendee'}</span></div>
        </div>
        <div class="card-details">
          ${a.email   ? `<div class="detail-wide"><div class="detail-label">Email</div><div class="detail-value">${a.email}</div></div>` : ''}
          ${a.contact ? `<div><div class="detail-label">Contact</div><div class="detail-value">${a.contact}</div></div>` : ''}
          ${a.guests > 0 ? `<div><div class="detail-label">Guests</div><div class="detail-value">${a.guests} (${a.guestNames || 'unnamed'})</div></div>` : ''}
          ${a.comments ? `<div class="detail-wide"><div class="detail-label">Comments</div><div class="detail-value">${a.comments}</div></div>` : ''}
          <div class="detail-wide"><div class="detail-label">Days</div><div class="detail-value">${a.days || '—'}</div></div>
        </div>
        <div class="card-actions">${actions}</div>
      </div>`;
  }

  function renderList(elId, list, dayContext) {
    const el = document.getElementById(elId);
    if (!list || list.length === 0) {
      el.innerHTML = '<div class="empty"><div class="empty-icon">&#10003;</div><div class="empty-text">Nothing here</div></div>';
      return;
    }
    el.innerHTML = list.map(a => cardHtml(a, dayContext)).join('');
  }

  // ---- SEARCH ----
  let searchDebounce;
  function handleSearchInput() {
    const q = document.getElementById('search-input').value.trim();
    document.getElementById('search-clear-btn').style.display = q ? 'inline-block' : 'none';
    clearTimeout(searchDebounce);
    if (!q) { searchMode = false; document.getElementById('search-banner-wrap').innerHTML = ''; renderCurrentTab(); return; }
    searchDebounce = setTimeout(() => runSearch(q), 250);
  }

  function runSearch(q) {
    searchMode = true;
    google.script.run.withSuccessHandler(function (results) {
      searchResults = results;
      document.getElementById('search-banner-wrap').innerHTML =
        `<div class="search-banner"><span>${results.length} result${results.length !== 1 ? 's' : ''} for "${q}"</span>
          <button class="search-clear-x" onclick="clearSearch()">&times;</button></div>`;
      renderSearchResults();
    }).withFailureHandler(handleError).searchAttendee(q);
  }

  function renderSearchResults() {
    renderList(currentTab === 'trash' ? 'all-list' : (currentTab + '-list'), searchResults, currentTab === 'all' ? 'all' : currentTab);
  }

  function clearSearch() {
    document.getElementById('search-input').value = '';
    document.getElementById('search-clear-btn').style.display = 'none';
    document.getElementById('search-banner-wrap').innerHTML = '';
    searchMode = false;
    renderCurrentTab();
  }

  // ---- QR SCANNER ----
  let scanStream = null;
  let scanRAF = null;

  function openScanModal() {
    document.getElementById('scan-modal').style.display = 'flex';
    const video = document.getElementById('scanVideo');
    navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } })
      .then(stream => {
        scanStream = stream;
        video.srcObject = stream;
        video.play();
        scanRAF = requestAnimationFrame(tickScan);
      })
      .catch(() => {
        document.getElementById('scan-status').textContent = 'Could not access the camera. Check permissions and try again, or search by name instead.';
      });
  }

  function closeScanModal() {
    document.getElementById('scan-modal').style.display = 'none';
    if (scanRAF) cancelAnimationFrame(scanRAF);
    if (scanStream) { scanStream.getTracks().forEach(t => t.stop()); scanStream = null; }
  }

  function tickScan() {
    const video = document.getElementById('scanVideo');
    const canvas = document.getElementById('scanCanvas');
    if (video.readyState === video.HAVE_ENOUGH_DATA) {
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
      const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
      const code = jsQR(imageData.data, imageData.width, imageData.height);
      if (code && code.data) {
        onScanSuccess(code.data);
        return;
      }
    }
    scanRAF = requestAnimationFrame(tickScan);
  }

  function onScanSuccess(text) {
    closeScanModal();
    const confNum = text.trim();
    // Determine day from active tab; default to day1
    const day = currentTab === 'day2' ? 'day2' : currentTab === 'workshop' ? 'workshop' : 'day1';
    showToast('Scanning ' + confNum + '...');
    google.script.run
      .withSuccessHandler(function(res) {
        if (!res.success) {
          showToast('Not found: ' + confNum);
          // Fall back to search so staff can see and handle manually
          document.getElementById('search-input').value = confNum;
          handleSearchInput();
        } else if (res.alreadyCheckedIn) {
          showToast('Already checked in: ' + res.name);
          loadAll(renderCurrentTab);
        } else {
          showToast('Checked in: ' + res.name);
          loadAll(renderCurrentTab);
        }
      })
      .withFailureHandler(function(err) {
        showToast('Scan error — searching instead');
        document.getElementById('search-input').value = confNum;
        handleSearchInput();
      })
      .quickCheckIn(confNum, day);
  }

  // ---- CHECK-IN ACTIONS ----
  function doCheckIn(rowIndex, day, btn) {
    btn.disabled = true; btn.textContent = 'Checking in...';
    google.script.run.withSuccessHandler(function () {
      showToast('Checked in');
      loadAll(renderCurrentTab);
    }).withFailureHandler(handleError).checkInAttendee(rowIndex, day);
  }

  function doUncheckIn(rowIndex, day, btn) {
    btn.disabled = true; btn.textContent = 'Undoing...';
    google.script.run.withSuccessHandler(function () {
      showToast('Check-in undone');
      loadAll(renderCurrentTab);
    }).withFailureHandler(handleError).uncheckInAttendee(rowIndex, day);
  }

  function doDelete(rowIndex, name, btn) {
    if (!confirm('Delete ' + name + '? This moves them to Trash, you can restore later.')) return;
    btn.disabled = true;
    google.script.run.withSuccessHandler(function () {
      showToast('Deleted ' + name);
      loadAll(renderCurrentTab);
    }).withFailureHandler(handleError).deleteAttendee(rowIndex);
  }

  function loadTrash() {
    document.getElementById('trash-list').innerHTML = '<div class="loading">Loading...</div>';
    google.script.run.withSuccessHandler(function (items) {
      if (!items || items.length === 0) {
        document.getElementById('trash-list').innerHTML = '<div class="empty"><div class="empty-icon">&#10003;</div><div class="empty-text">Trash is empty</div></div>';
        return;
      }
      document.getElementById('trash-list').innerHTML = items.map(a => `
        <div class="result-card" id="trash-card-${a.rowIndex}" style="opacity:0.75;">
          <div class="card-top">
            <div>
              <div class="card-name">${a.name}</div>
              ${a.confirmNum ? `<div class="card-conf">${a.confirmNum}</div>` : ''}
            </div>
            <span class="badge badge-role">${a.role || 'Attendee'}</span>
          </div>
          <div class="card-actions">
            <button class="btn btn-teal btn-sm" onclick="doRestore(${a.rowIndex},'${(a.name||'').replace(/'/g,"\\'")}',this)">Restore</button>
          </div>
        </div>`).join('');
    }).withFailureHandler(handleError).getDeletedAttendees();
  }

  function doRestore(rowIndex, name, btn) {
    btn.disabled = true; btn.textContent = 'Restoring...';
    google.script.run.withSuccessHandler(function () {
      const el = document.getElementById('trash-card-' + rowIndex);
      if (el) el.remove();
      showToast('Restored ' + name);
      loadAll(renderCurrentTab);
    }).withFailureHandler(handleError).restoreAttendee(rowIndex);
  }

  // ---- WALK-IN REGISTRATION ----
  function openRegisterModal() {
    document.getElementById('register-modal').style.display = 'flex';
    document.getElementById('register-modal-btns').style.display = 'flex';
    document.getElementById('reg-confirm-display').style.display = 'none';
    ['reg-name','reg-contact','reg-email','reg-guest-names'].forEach(id => document.getElementById(id).value = '');
    document.getElementById('reg-guests').value = 0;
    ['reg-day1','reg-day2','reg-checkin-day1','reg-checkin-day2'].forEach(id => document.getElementById(id).checked = false);
  }

  function closeRegisterModal() { document.getElementById('register-modal').style.display = 'none'; }

  function submitRegistration() {
    const name = document.getElementById('reg-name').value.trim();
    if (!name) { showToast('Name is required'); return; }
    const day1sel = document.getElementById('reg-day1').checked;
    const day2sel = document.getElementById('reg-day2').checked;
    if (!day1sel && !day2sel) { showToast('Select at least one day'); return; }

    const dayParts = [];
    if (day1sel) dayParts.push('Day 1 - July 13, 2026 (Lozova)');
    if (day2sel) dayParts.push('Day 2 - July 14, 2026 (Scoreni)');

    const regBtn = document.querySelector('#register-modal-btns .btn-teal');
    regBtn.disabled = true; regBtn.textContent = 'Registering...';

    const data = {
      name: name,
      role: document.getElementById('reg-role').value,
      days: dayParts.join(', '),
      guestCount: parseInt(document.getElementById('reg-guests').value) || 0,
      guestNames: document.getElementById('reg-guest-names').value.trim(),
      contact: document.getElementById('reg-contact').value.trim(),
      email: document.getElementById('reg-email').value.trim(),
      checkInDay1: document.getElementById('reg-checkin-day1').checked,
      checkInDay2: document.getElementById('reg-checkin-day2').checked,
    };

    google.script.run
      .withSuccessHandler(function (res) {
        document.getElementById('register-modal-btns').style.display = 'none';
        document.getElementById('reg-confirm-display').style.display = 'block';
        document.getElementById('reg-confirm-num').textContent = res.confirmNum;
        showToast('Registered: ' + name);
        loadAll(renderCurrentTab);
      })
      .withFailureHandler(function (err) {
        regBtn.disabled = false; regBtn.textContent = 'Register';
        handleError(err);
      })
      .registerWalkin(data);
  }

  function confirmResendAll() {
    const withEmail = allData.all.filter(a => a.email).length;
    const total = allData.all.length;
    if (!confirm(`Resend summit passes to all ${withEmail} attendees with email addresses?\n(${total - withEmail} will be skipped — no email on file)\n\nThis uses ${withEmail} of your daily Gmail quota.`)) return;
    showToast('Sending ' + withEmail + ' passes...');
    google.script.run
      .withSuccessHandler(function(res) { showToast('Sent ' + res.sent + ' passes, skipped ' + res.skipped); })
      .withFailureHandler(handleError)
      .resendAllConfirmationEmails();
  }

  function doResend(rowIndex, btn) {
    btn.disabled = true; btn.textContent = 'Sending...';
    google.script.run
      .withSuccessHandler(function() { btn.disabled = false; btn.textContent = '📧 Resend Pass'; showToast('Pass email sent!'); })
      .withFailureHandler(function(err) { btn.disabled = false; btn.textContent = '📧 Resend Pass'; handleError(err); })
      .resendConfirmationEmail(rowIndex);
  }

  // ---- UTILS ----
  function showToast(msg) {
    const t = document.getElementById('toast');
    t.textContent = msg; t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), 2500);
  }
  function handleError(err) { showToast('Error: ' + (err.message || String(err))); }

  setInterval(() => { loadAll(renderCurrentTab); }, 30000);
  loadAll(renderCurrentTab);
